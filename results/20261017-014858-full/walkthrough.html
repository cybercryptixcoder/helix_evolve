import numpy as np

def feature(item, bins, capacity):
    """Residual slack after placement: clip(min residual,0,1) minus 0.5 if too tight (<0.2)."""
    cap = np.asarray(capacity, dtype=float)
    r = (bins - item) / cap
    m = np.minimum(r[:, 0], r[:, 1])
    slack = np.clip(m, 0.0, 1.0)
    tight = 0.5 * (m < 0.2)
    return slack - tight

import numpy as np
import math

def feature(item, bins, capacity):
    """Negative of the max leftover fraction of the tighter resource after placing: prefers balanced-fit bins."""
    cap = np.maximum(np.asarray(capacity, dtype=float), 1.0)
    b = np.asarray(bins, dtype=float)
    it = np.asarray(item, dtype=float)
    rem = b - it  # remaining after placement, per resource
    # fraction of capacity left in each resource after placing
    frac = rem / cap[None, :]
    # tighter leftover = min over resources (worst-case slack); negative so that smaller slack scores higher
    tighter = np.minimum(frac[:, 0], frac[:, 1])

residual_imbalance_times_fullness: code
import numpy as np

def feature(item, bins, capacity):
    """Relative imbalance of what remains after placing the job, scaled by bin fullness."""
    cap = np.maximum(np.asarray(capacity, dtype=float), 1.0)
    rem = np.asarray(bins, dtype=float) - np.asarray(item, dtype=float)
    rem = np.clip(rem, 0.0, None)
    r = rem / cap
    # imbalance between the two residual resources, signed by which side is tighter
    imbalance = (r[:, 0] - r[:, 1]) / (r[:, 0] + r[:, 1] + 1e-9)
    # how full the bin would be overall (0 = empty, 1 = both resources exhausted)
    fullness = 1.0 - 0.5 * (r[:, 0] + r[:, 1])
    return imbalance * fullness

def feature(item, bins, capacity):
    """Cosine of the angle between the bin's remaining-capacity vector and the item's demand vector."""
    cap = np.asarray(capacity, dtype=float)
    it = np.asarray(item, dtype=float)

    """Best-fit slack after placement: geometric mean of remaining fractions, job-normalised."""
    cap = np.maximum(np.asarray(capacity, dtype=float), 1.0)

import numpy as np


def feature(item, bins, capacity):
    """How much of the bin's *smaller* remaining resource is consumed, relative to its larger remaining one."""
    cap = np.asarray(capacity, dtype=float)
    b = np.asarray(bins, dtype=float)
    # remaining fractions in each resource
    frac = np.clip(b / np.maximum(cap, 1.0), 0.0, 1.0)
    lo = frac.min(axis=1)
    hi = frac.max(axis=1)
    # after placing the item, the bottleneck resource of the bin
    rem = np.clip(b - np.asarray(item, dtype=float), 0.0, None)
    rem_frac = rem / np.maximum(cap, 1.0)
    new_lo = rem_frac.min(axis=1)
    # balance measure: prefer bins whose two resources drain together
    return (new_lo - lo) + (hi - lo) * 0.5

def feature(item, bins, capacity):
    """Post-placement residual imbalance, rewarded only when the tight side keeps >5% room."""
    cap = np.asarray(capacity, dtype=float)

import numpy as np

def feature(item, bins, capacity):
    """Active for large items: reward tight post-placement residual that is very small (bin closes) or very large (room for another large), penalise the middle band."""
    cap = np.asarray(capacity, dtype=float)
    it = np.asarray(item, dtype=float)
    r = (bins.astype(float) - it) / cap
    tight = np.minimum(r[:, 0], r[:, 1])
    active = (np.max(it / cap) > 0.4)
    val = -tight * (tight > 0.2) + tight * (tight <= 0.2)
    return (val * active).astype(float)

def feature(item, bins, capacity):
    """Asymmetric post-placement residual: reward bins where the *binding* resource is nearly full while the other stays slack, penalise the reverse."""
    cap = np.asarray(capacity, dtype=float)
    it = np.asarray(item, dtype=float)
    r = (bins - it) / cap                      # post-placement residual, per resource
    tight = np.minimum(r[:, 0], r[:, 1])       # binding resource after placement
    loose = np.maximum(r[:, 0], r[:, 1])       # slack resource after placement
    # A bin is "well shaped" if its binding resource is close to full (tight small)
    # and it still has room elsewhere (loose large). Reward that shape.
    # Weight the tightness more strongly than plain mean, and only reward the good side.
    shaped = (loose - tight)                   # spread in [0,1], larger = more lopsided
    fill = 1.0 - tight                         # how full the binding resource becomes
    # Reward lopsided bins that are full on their binding side; punish lopsided bins
    # that waste a resource (binding side slack while other side full is impossible,
    # so this reduces to: prefer filling the binding resource).
    val = fill - 0.5 * shaped
    # Down-weight the empty (last) row's contribution to keep neutral baseline.
    return np.clip(val, -1.0, 1.0)

def feature(item, bins, capacity):
    """Asymmetric stranding penalty: only penalise when the tight residual is small AND the loose one large, weighted by how imbalanced the item itself is."""
    cap = np.asarray(capacity, dtype=float)
    it = np.asarray(item, dtype=float)
    r = (bins - it) / cap
    m = np.minimum(r[:, 0], r[:, 1])
    M = np.maximum(r[:, 0], r[:, 1])

    # how lopsided the arriving job is: 0 = balanced, 1 = purely one resource
    ic = it / cap
    imb = np.abs(ic[0] - ic[1]) / (ic[0] + ic[1] + 1e-9)

    tight_small = np.clip((0.10 - m) / 0.10, 0.0, 1.0)
    loose_big = np.clip((M - 0.20) / 0.30, 0.0, 1.0)

    # only meaningful for jobs that themselves consume mostly one resource

The Mutator wrote 20 features: 16 were rejected by the probe, 2 helped but were redundant (they repeated what the kept features already did), 1 was refuted and 1 was flat. In this run, the wide loop didn't find anything the deep loop then built on, which is the opposite of the "get lucky, then cash in" story.

The wide loop isn't wide

The Mutator's variant requests are aimed at the champion's own features, ordered by importance. With an empty library (generations 1 and 2) it can only write fresh features, and once one feature existed, 7 of its 9 variants over the next three generations were variants of that same feature (class_match_residual, f0013). It has no memory between generations (the air gap also keeps the research log away from it), so it re-proposed a cosine-alignment feature it had already tried: f0004 in generation 1 and f0009 in generation 2 are the same feature, with identical probe results (−14.257 and −26.292).

The same idea comes back under a new name, and nothing notices
  • Generation 1: f0001 (Mutator) and f0007 (Researcher) are the same measurement, the tighter resource's leftover after placing, and score identically (−0.350). f0005 ranks bins the same way with the sign flipped, and the probe tries both signs, so it scored −0.350 too. Three of eight slots went on one idea.
  • f0004 and f0009: the same cosine feature, a generation apart.
  • Generation 4's anticorr_fourth_slot (f0032) is generation 2's pair_complementarity (f0014) with the sign flipped.
  • Generation 3's open_bin_fill_preference (f0023) is generation 1's f0007 plus a fixed penalty on the new bin.

The research log shows verdicts, but nothing records that two features are the same approach. The probe results are a free fingerprint for this: two features with the same probe scores on the same champion rank bins the same way.

The probe decides most features' fate, one feature at a time

30 of the 40 features were rejected by the four-weight probe before any refitting. Each is judged alone on top of the champion, so a feature that only helps next to another one (a tie-breaker, say) can't survive. The mechanisms that would retry those, pair trials and the backlog revisit, are off by default (--pairs 0, --revisit 0). The search is greedy: one coordinate at a time, kept or dropped.

What gets credited is the formula, not the idea
  • f0013 is named and argued as class matching (does the job's size class complement the bin's leftover shape?), but the measurement written for it is a pure stranding penalty: −(larger leftover − smaller leftover) when the smaller is under 5% and the larger over 25%. The stranding part is what was validated (+1.513), and the log records the class-matching claim as supported.
  • f0037's claim says to reward a large job's bin when it ends very tight or very roomy, and penalise the middle. Its formula rewards tight and penalises roomy, the Coder said so and implemented the formula literally, and the optimiser then fitted a negative weight. So the kept behaviour is the reverse of the formula: for large jobs, prefer bins that keep more than 20% room in both resources. Half of the claim, reached by accident.

The Researcher reads these verdicts next generation as evidence about its claims. Nothing checks that the claim, the measurement and the fitted sign agree.

Textbook Best Fit only arrived in generation 4

closing_bin_penalty (f0031), −(mean leftover after placing), is Best Fit. By the end it's the feature the policy relies on most (importance 5.72, against 1.63 for the stranding penalty). In generation 1 both loops proposed a different best fit, on the tighter resource only, which is worse than first fit here (−0.350) because it fills one resource and strands the other.

"Stuck" is declared after one generation

In generations 2 and 4 the Researcher was told "the policy hasn't improved for 1 generations: the small refinements are used up". In generation 2 the policy was still the empty Code Zero, so there were no refinements to use up. The message fires when stuck >= 1, while the extra ideas and the extra Mutator slot start at stuck >= 2.

There is very little reasoning to see

Hidden thinking is switched off (reasoning_effort: none), so what you read is all the reasoning there is. The Researcher writes its JSON first, then at most 100 words, in about 8 s. The Coder writes code and one sentence. The Mutator writes code and a name, with no explanation at all. The Researcher's claims are the only place the system says why. All thinking levels answer again on Luminal (checked today), so trying thinking for the Researcher alone is now possible and cheap.

This isn't the "exaggerate one aspect" design

The deep loop doesn't write exaggerated variants of the champion. It proposes a new measurement and Fit credits it with the difference it makes, with versus without. The exaggeration probes were the first design and were dropped after runs 1 and 2: the distorted copies always scored worse and crowded the pool (HANDOFF.md §9). If you want probes that exaggerate one aspect to learn about it, that's a change to build, not something the current code does.

Fitting time grows when things go well

Generations 1 to 4 took 13-32 s. Generation 5 took 152 s, 150 of them fitting: five features passed validation, so combining them, the joint refit and two second-chance retries all ran. The agents' time stayed at about 10 s per generation throughout.
